cmake_minimum_required(VERSION 3.27...4.3)
project(alps_sdk_contract LANGUAGES C CXX)

# A parent may already use MPI and have its own numerical-provider preferences.
set(MPI_FOUND TRUE)
set(BLA_VENDOR ParentProvider)
set(BLA_STATIC OFF)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -DCONSUMER_FLAG=1")
set(saved_flags "${CMAKE_CXX_FLAGS}")
if(ALPS_SOURCE)
  unset(BLA_VENDOR)
  add_subdirectory("${ALPS_SOURCE}" alps EXCLUDE_FROM_ALL)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:ALPS::alps>\n")
  if(ALPS_ENABLE_MPI
     OR (ALPS_BUILD_APPLICATIONS AND NOT EXPECT_SOLVERS)
     OR ALPS_BUILD_TESTING)
    message(FATAL_ERROR "Embedded ALPS must default to a library-only build")
  endif()
  foreach(property IN ITEMS POSITION_INDEPENDENT_CODE RUNTIME_OUTPUT_DIRECTORY
                            LIBRARY_OUTPUT_DIRECTORY ARCHIVE_OUTPUT_DIRECTORY)
    get_target_property(actual alps ${property})
    if(NOT "${actual}" STREQUAL "${CMAKE_${property}}")
      message(FATAL_ERROR "Embedded ALPS ignored CMAKE_${property}: ${actual}")
    endif()
  endforeach()
else()
  if(EXPECT_MISSING_DEPENDENCY)
    find_package(ALPS CONFIG QUIET)
    if(ALPS_FOUND OR NOT ALPS_NOT_FOUND_MESSAGE MATCHES "Boost" OR TARGET ALPS::alps)
      message(FATAL_ERROR "The SDK lost the missing dependency diagnostic: ${ALPS_NOT_FOUND_MESSAGE}")
    endif()
    return()
  endif()
  if(EXPECT_APPLICATIONS)
    find_package(ALPS CONFIG REQUIRED COMPONENTS applications utilities hdf5 params osiris xml cli)
    set(application_paths "")
    foreach(application IN LISTS ALPS_APPLICATION_TARGETS)
      string(APPEND application_paths "$<TARGET_FILE:${application}>\n")
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/applications-$<CONFIG>.txt"
      CONTENT "${application_paths}")
  else()
    find_package(ALPS CONFIG REQUIRED COMPONENTS utilities hdf5 params osiris xml cli numerics numeric_io)
  endif()
  if(EXPECT_SOLVERS)
    find_package(ALPS CONFIG REQUIRED COMPONENTS solvers)
  endif()
  get_filename_component(expected_prefix "${ALPS_DIR}/../.." REALPATH)
  if(EXISTS "${expected_prefix}/include/alps/ngs/detail/export_sim_to_python.hpp"
     OR EXISTS "${expected_prefix}/include/alps/python/save_observable_to_hdf5.hpp")
    message(FATAL_ERROR "The C++ SDK still installs Python-owned headers")
  endif()
  foreach(private_directory IN ITEMS alps/utilities alps/hdf5/include alps/hdf5/src
                                     alps/hdf5/tests alps/params alps/common alps/runtime
                                     alps/fortran/include alps/fortran/src alps/fortran/tests)
    if(EXISTS "${expected_prefix}/include/${private_directory}")
      message(FATAL_ERROR "The SDK installs a module's private layout: ${private_directory}")
    endif()
  endforeach()
  cmake_path(IS_PREFIX expected_prefix "${ALPS_DATA_DIR}" NORMALIZE data_relocated)
  if(NOT data_relocated OR COMMAND alps_target_link_pyalps)
    message(FATAL_ERROR "The SDK leaked dependency paths or Python integration")
  endif()
  get_target_property(runtime ALPS::alps IMPORTED_LOCATION_RELEASE)
  cmake_path(IS_PREFIX expected_prefix "${runtime}" NORMALIZE relocated)
  if(NOT relocated)
    message(FATAL_ERROR "The SDK runtime points outside its installed prefix: ${runtime}")
  endif()
  if(NOT BLA_VENDOR STREQUAL "ParentProvider" OR NOT BLA_STATIC STREQUAL "OFF")
    message(FATAL_ERROR "The SDK changed the parent's numerical-provider options")
  endif()
  if(DEFINED BLA_SIZEOF_INTEGER AND NOT EXPECT_ABI)
    message(FATAL_ERROR "The SDK leaked BLA_SIZEOF_INTEGER into its parent")
  endif()
endif()
if(NOT CMAKE_CXX_FLAGS STREQUAL saved_flags)
  message(FATAL_ERROR "ALPS changed the parent's compiler flags")
endif()
get_target_property(links ALPS::alps INTERFACE_LINK_LIBRARIES)
if(NOT TARGET ALPS::fortran)
  message(FATAL_ERROR "The SDK must always supply the C++ Fortran bridge")
endif()
if(links MATCHES "MPI::")
  message(FATAL_ERROR "MPI leaked into an MPI-disabled ALPS target")
endif()
get_target_property(options ALPS::alps INTERFACE_COMPILE_OPTIONS)
if(options MATCHES "bigobj|permissive")
  message(FATAL_ERROR "ALPS exported private compiler switches")
endif()
add_executable(sdk_contract main.cpp)
target_link_libraries(sdk_contract PRIVATE ALPS::alps)
add_executable(numeric_contract numerics.cpp)
target_link_libraries(numeric_contract PRIVATE ALPS::numerics)
get_target_property(utility_type ALPS::utilities TYPE)
get_target_property(utility_links ALPS::utilities INTERFACE_LINK_LIBRARIES)
if(NOT utility_type MATCHES "^(SHARED|STATIC)_LIBRARY$"
   OR utility_links MATCHES "ALPS::alps|HDF5::|MPI::|[Bb][Ll][Aa][Ss]|[Ll][Aa][Pp][Aa][Cc][Kk]")
  message(FATAL_ERROR "Utilities must be a real library independent of the scientific runtime")
endif()
add_executable(numeric_io_contract numeric_io.cpp)
target_link_libraries(numeric_io_contract PRIVATE ALPS::numeric_io)
add_executable(utilities_contract utilities.cpp)
target_link_libraries(utilities_contract PRIVATE ALPS::utilities)
get_target_property(hdf5_type ALPS::hdf5 TYPE)
get_target_property(hdf5_links ALPS::hdf5 INTERFACE_LINK_LIBRARIES)
if(NOT hdf5_type MATCHES "^(SHARED|STATIC)_LIBRARY$"
   OR hdf5_links MATCHES "ALPS::alps|MPI::|[Bb][Ll][Aa][Ss]|[Ll][Aa][Pp][Aa][Cc][Kk]")
  message(FATAL_ERROR "HDF5 must be a real library independent of the simulation runtime")
endif()
add_executable(hdf5_contract hdf5.cpp)
target_link_libraries(hdf5_contract PRIVATE ALPS::hdf5)
get_target_property(params_type ALPS::params TYPE)
get_target_property(params_links ALPS::params INTERFACE_LINK_LIBRARIES)
if(NOT params_type MATCHES "^(SHARED|STATIC)_LIBRARY$"
   OR params_links MATCHES "ALPS::alps|MPI::|[Bb][Ll][Aa][Ss]|[Ll][Aa][Pp][Aa][Cc][Kk]")
  message(FATAL_ERROR "Typed params must be a real library independent of the simulation runtime")
endif()
add_executable(params_contract params.cpp)
target_link_libraries(params_contract PRIVATE ALPS::params)
get_target_property(osiris_type ALPS::osiris TYPE)
get_target_property(osiris_links ALPS::osiris INTERFACE_LINK_LIBRARIES)
if(NOT osiris_type MATCHES "^(SHARED|STATIC)_LIBRARY$"
   OR osiris_links MATCHES "ALPS::alps|HDF5::|[Bb][Ll][Aa][Ss]|[Ll][Aa][Pp][Aa][Cc][Kk]")
  message(FATAL_ERROR "Osiris must be a real library independent of the simulation runtime")
endif()
add_executable(osiris_contract ../../../src/alps/osiris/tests/comm_contract.cpp)
target_link_libraries(osiris_contract PRIVATE ALPS::osiris)
foreach(component IN ITEMS xml cli)
  get_target_property(component_type ALPS::${component} TYPE)
  get_target_property(component_links ALPS::${component} INTERFACE_LINK_LIBRARIES)
  if(NOT component_type MATCHES "^(SHARED|STATIC)_LIBRARY$"
     OR component_links MATCHES "ALPS::alps|ALPS::params|ALPS::hdf5|HDF5::|MPI::|[Bb][Ll][Aa][Ss]|[Ll][Aa][Pp][Aa][Cc][Kk]")
    message(FATAL_ERROR "${component} must link independently of archive/simulation libraries")
  endif()
endforeach()
add_executable(xml_contract xml.cpp)
target_link_libraries(xml_contract PRIVATE ALPS::xml)
add_executable(cli_contract cli.cpp)
target_link_libraries(cli_contract PRIVATE ALPS::cli)
add_executable(parameter_adapters_contract ../../integration/params/legacy_adapters.cpp)
target_link_libraries(parameter_adapters_contract PRIVATE ALPS::alps)
enable_testing()
add_test(NAME numeric_contract COMMAND numeric_contract)
add_test(NAME numeric_io_contract COMMAND numeric_io_contract)
add_test(NAME utilities_contract COMMAND utilities_contract)
add_test(NAME hdf5_contract COMMAND hdf5_contract)
add_test(NAME params_contract COMMAND params_contract)
add_test(NAME osiris_contract COMMAND osiris_contract)
add_test(NAME xml_contract COMMAND xml_contract)
add_test(NAME cli_contract COMMAND cli_contract)
add_test(NAME parameter_adapters_contract COMMAND parameter_adapters_contract)
if(EXPECT_SOLVERS)
  get_target_property(maxent_links ALPS::maxent INTERFACE_LINK_LIBRARIES)
  if(maxent_links MATCHES "(^|;)ALPS::alps(;|$)")
    message(FATAL_ERROR "MaxEnt must link independently of the simulation runtime")
  endif()
  add_executable(maxent_independent_contract maxent_independent.cpp)
  target_link_libraries(maxent_independent_contract PRIVATE ALPS::maxent)
  add_test(NAME maxent_independent_contract COMMAND maxent_independent_contract)
  add_executable(solver_contract solvers.cpp)
  target_link_libraries(solver_contract PRIVATE ALPS::maxent ALPS::cthyb ALPS::ctint)
  add_test(NAME solver_contract COMMAND solver_contract)
  set_tests_properties(solver_contract maxent_independent_contract PROPERTIES ENVIRONMENT_MODIFICATION
    "PATH=path_list_prepend:${ALPS_RUNTIME_LIBRARY_DIRS}")
endif()
add_test(NAME sdk_contract COMMAND sdk_contract)
if(NOT ALPS_SOURCE)
  add_test(NAME sdk_resources COMMAND "${CMAKE_COMMAND}" -E env --unset=ALPS_XML_PATH
    "ALPS_ROOT=${expected_prefix}" -- "$<TARGET_FILE:sdk_contract>" "${ALPS_DATA_DIR}/xml")
endif()
if(NOT ALPS_SOURCE)
  set_tests_properties(sdk_contract sdk_resources numeric_contract numeric_io_contract utilities_contract hdf5_contract
    params_contract osiris_contract xml_contract cli_contract parameter_adapters_contract PROPERTIES ENVIRONMENT_MODIFICATION
    "PATH=path_list_prepend:${ALPS_RUNTIME_LIBRARY_DIRS}")
endif()

if(ALPS_SOURCE)
  set(component_includes "")
  foreach(component IN ITEMS utilities hdf5 params osiris xml cli numeric numeric_io)
    string(APPEND component_includes
      "${component}=$<JOIN:$<TARGET_PROPERTY:${component}_contract,INCLUDE_DIRECTORIES>,|>\n")
  endforeach()
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/component-includes-$<CONFIG>.txt"
    CONTENT "${component_includes}")
endif()
