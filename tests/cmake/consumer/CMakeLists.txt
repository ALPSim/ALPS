# A parent may already use MPI and have its own numerical-provider preferences.
set(MPI_FOUND TRUE)

set(BLA_STATIC OFF)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -DCONSUMER_FLAG=1")
set(saved_flags "${CMAKE_CXX_FLAGS}")
if(ALPS_SOURCE)
  unset(BLA_VENDOR)
  add_subdirectory("${ALPS_SOURCE}" alps EXCLUDE_FROM_ALL)
  set(sdk_mpi_enabled "${ALPS_ENABLE_MPI}")
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:ALPS::alps>\n")
  if(ALPS_ENABLE_MPI
     OR (ALPS_BUILD_APPLICATIONS AND NOT EXPECT_SOLVERS)
     OR ALPS_BUILD_TESTING)
    message(FATAL_ERROR "Embedded ALPS must default to a library-only build")
  endif()
  foreach(property IN ITEMS POSITION_INDEPENDENT_CODE RUNTIME_OUTPUT_DIRECTORY
                            LIBRARY_OUTPUT_DIRECTORY ARCHIVE_OUTPUT_DIRECTORY)
    get_target_property(actual alps ${property})
    if(NOT "${actual}" STREQUAL "${CMAKE_${property}}")
      message(FATAL_ERROR "Embedded ALPS ignored CMAKE_${property}: ${actual}")
    endif()
  endforeach()
else()
  if(EXPECT_MISSING_DEPENDENCY)
    find_package(ALPS CONFIG QUIET)
    if(ALPS_FOUND OR NOT ALPS_NOT_FOUND_MESSAGE MATCHES "Boost" OR TARGET ALPS::alps)
      message(FATAL_ERROR "The SDK lost the missing dependency diagnostic: ${ALPS_NOT_FOUND_MESSAGE}")
    endif()
    return()
  endif()
  if(EXPECT_APPLICATIONS)
    find_package(ALPS CONFIG REQUIRED COMPONENTS applications utilities hdf5 params osiris xml cli)
    set(application_paths "")
    foreach(application IN LISTS ALPS_APPLICATION_TARGETS)
      string(APPEND application_paths "$<TARGET_FILE:${application}>\n")
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/applications-$<CONFIG>.txt"
      CONTENT "${application_paths}")
  else()
    find_package(ALPS CONFIG REQUIRED COMPONENTS utilities hdf5 params osiris xml cli numerics numeric_io)
  endif()
  if(EXPECT_SOLVERS)
    find_package(ALPS CONFIG REQUIRED COMPONENTS solvers)
  endif()
  set(sdk_mpi_enabled "${ALPS_MPI_ENABLED}")
  get_filename_component(expected_prefix "${ALPS_DIR}/../.." REALPATH)
  if(EXISTS "${expected_prefix}/include/alps/ngs/detail/export_sim_to_python.hpp"
     OR EXISTS "${expected_prefix}/include/alps/python/save_observable_to_hdf5.hpp")
    message(FATAL_ERROR "The C++ SDK still installs Python-owned headers")
  endif()
  foreach(private_directory IN ITEMS alps/utilities alps/hdf5/include alps/hdf5/src
                                     alps/hdf5/tests alps/params alps/common alps/runtime
                                     alps/fortran/include alps/fortran/src alps/fortran/tests)
    if(EXISTS "${expected_prefix}/include/${private_directory}")
      message(FATAL_ERROR "The SDK installs a module's private layout: ${private_directory}")
    endif()
  endforeach()
  cmake_path(IS_PREFIX expected_prefix "${ALPS_DATA_DIR}" NORMALIZE data_relocated)
  if(NOT data_relocated OR COMMAND alps_target_link_pyalps)
    message(FATAL_ERROR "The SDK leaked dependency paths or Python integration")
  endif()
  get_target_property(runtime ALPS::alps IMPORTED_LOCATION_RELEASE)
  cmake_path(IS_PREFIX expected_prefix "${runtime}" NORMALIZE relocated)
  if(NOT relocated)
    message(FATAL_ERROR "The SDK runtime points outside its installed prefix: ${runtime}")
  endif()
  if(NOT BLA_VENDOR STREQUAL "ParentProvider" OR NOT BLA_STATIC STREQUAL "OFF")
    message(FATAL_ERROR "The SDK changed the parent's numerical-provider options")
  endif()
  if(DEFINED BLA_SIZEOF_INTEGER AND NOT EXPECT_ABI)
    message(FATAL_ERROR "The SDK leaked BLA_SIZEOF_INTEGER into its parent")
  endif()
endif()
if(NOT CMAKE_CXX_FLAGS STREQUAL saved_flags)
  message(FATAL_ERROR "ALPS changed the parent's compiler flags")
endif()
get_target_property(links ALPS::alps INTERFACE_LINK_LIBRARIES)
if(NOT TARGET ALPS::fortran)
  message(FATAL_ERROR "The SDK must always supply the C++ Fortran bridge")
endif()
set(serial_dependencies "")
if(sdk_mpi_enabled)
  if(NOT links MATCHES "MPI::")
    message(FATAL_ERROR "The MPI-enabled SDK lost its MPI dependency")
  endif()
elseif(links MATCHES "MPI::")
  message(FATAL_ERROR "MPI leaked into an MPI-disabled ALPS target")
endif()
if(NOT sdk_mpi_enabled)
  set(serial_dependencies "|MPI::")
endif()
get_target_property(options ALPS::alps INTERFACE_COMPILE_OPTIONS)
if(options MATCHES "bigobj|permissive")
  message(FATAL_ERROR "ALPS exported private compiler switches")
endif()
enable_testing()
function(add_contract name source target)
  add_executable(${name}_contract ${source})
  # These consumers use headers, not modules. GCC's module mode is incompatible
  # with some Apple SDK headers even when no module is imported.
  set_target_properties(${name}_contract PROPERTIES CXX_SCAN_FOR_MODULES OFF)
  target_link_libraries(${name}_contract PRIVATE ${target})
  add_test(NAME ${name}_contract COMMAND ${name}_contract)
endfunction()

function(add_component_contract component source forbidden)
  get_target_property(type ALPS::${component} TYPE)
  get_target_property(links ALPS::${component} INTERFACE_LINK_LIBRARIES)
  if(NOT type MATCHES "^(SHARED|STATIC)_LIBRARY$" OR links MATCHES "${forbidden}")
    message(FATAL_ERROR "${component} must be a real library with independent dependencies: ${links}")
  endif()
  add_contract(${component} ${source} ALPS::${component})
endfunction()
